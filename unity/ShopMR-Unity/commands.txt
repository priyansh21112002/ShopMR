# 2. Start session
curl -s -X POST http://localhost:8000/api/sessions/start \
  -H "Content-Type: application/json" \
  -d '{"device_id": "test"}' | python3 -m json.tool

# 3. Chat (replace SESSION_ID)
curl -s -X POST http://localhost:8000/api/chat/message \
  -H "Content-Type: application/json" \
  -d '{"session_id": "SESSION_ID", "message": "Hi", "conversation_history": []}' \
  | python3 -m json.tool

# 4. List products
curl -s http://localhost:8000/api/products/ | python3 -m json.tool
